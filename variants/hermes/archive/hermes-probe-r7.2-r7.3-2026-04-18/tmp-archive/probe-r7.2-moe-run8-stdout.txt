```python
import os

def count_large_files(directory, size_threshold_mb):
    threshold_bytes = size_threshold_mb * 1024 * 1024
    count = 0
    
    # Expand user to handle '~'
    expanded_dir = os.path.expanduser(directory)
    
    if not os.path.exists(expanded_dir):
        print(f"Error: Directory {expanded_dir} does not exist.")
        return

    for root, dirs, files in os.walk(expanded_dir):
        for name in files:
            file_path = os.path.join(root, name)
            try:
                # Use os.path.getsize to check file size
                # Follow symlinks? Default is False, which is safer
                if os.path.getsize(file_path) > threshold_bytes:
                    count += 1
            except (OSError, PermissionError):
                # Skip files that can't be accessed (e.g. permission denied)
                continue
                
    print(f"Total files larger than {size_threshold_mb}MB in {directory}: {count}")

if __name__ == "__main__":
    count_large_files("~/Downloads", 10)
```

I've written this as a clean, standalone script. It uses `os.walk` to ensure it checks subdirectories within your Downloads folder and includes basic error handling to skip files it doesn't have permission to read.
